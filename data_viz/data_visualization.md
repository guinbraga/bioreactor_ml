```python
from sklearn.preprocessing import FunctionTransformer
from skbio.stats.composition import multi_replace
from optuna.trial import Trial, FixedTrial
from pandas import DataFrame
def clr_transform(X: np.ndarray | DataFrame) -> np.ndarray | DataFrame:
    """
    Implements the Centered Log Ratio transform, which is recommended for
    dealing with compositional data as such generated by High Throughput
    Sequencing
    """
    # Need to make sure X is at least 2d to not have problems with LeaveOneOut
    X_zeroless = multi_replace(np.asarray(X))
    X_2d = np.atleast_2d(X_zeroless)

    log_X = np.log(X_2d)
    geo_mean = log_X.mean(axis=1, keepdims=True)

    return log_X - geo_mean


class CLRTransformer:
    def create_scaler(self, trial: Trial | FixedTrial):
        return FunctionTransformer(clr_transform, feature_names_out='one-to-one')
asv_all = pd.read_csv("../../04_bioreactor_ml_project/data/genomic/03_map_complete_absolute_n_hits_table.csv", index_col=0).T
```

#### Matplotlib

```python
import matplotlib.pyplot as plt
import matplotlib.lines as mlines
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

# 1. Prepare your data (Assuming pca, pca_results, and exp_var are already calculated)
# Extract the binary features from the 'TRANSFER' column
metadata['Timepoint'] = metadata['TRANSFER'].apply(lambda x: x.split('-')[0]) # Yields '1T', '3T'
metadata['Procedure'] = metadata['TRANSFER'].apply(lambda x: x.split('-')[1]) # Yields '1', '2'

# 2. Define the 3-Way Aesthetic Dictionaries
color_map = {'Midgut': '#1f77b4', 'Hindgut': '#ff7f0e'}  # Blue and Orange
shape_map = {'1T': 'o', '3T': '^'}                       # Circle and Triangle

# 3. Setup Figure
fig, ax = plt.subplots(figsize=(12, 7))

# 4. Group data and plot iterate through the 3 dimensions
# Grouping by the three binary categories guarantees we map every unique combination correctly
for (compartment, time, cond), group_data in metadata.groupby(['Gut Compartment', 'Timepoint', 'Procedure']):
    
    # Map color and shape
    base_color = color_map[compartment]
    marker_shape = shape_map[time]
    
    # Map Fill: If '-1', filled (facecolor = base_color). If '-2', unfilled (facecolor = 'none')
    if cond == 'P1':
        face_color = base_color
        edge_width = 0.5
    else:
        face_color = 'none'
        edge_width = 2.0  # Thicker edge to make unfilled shapes pop
        
    ax.scatter(
        group_data['PC1'], 
        group_data['PC2'],
        c=face_color,
        edgecolors=base_color,
        marker=marker_shape,
        linewidths=edge_width,
        s=150,           # Slightly larger to make hollow shapes easier to see
        alpha=0.85,
        label=f"{compartment} {time}-{cond}" # Optional, mostly useful for debugging here
    )

# 5. Styling and Layout Customization
ax.set_title('PCA of CLR-Transformed Microbial Distribution', fontsize=16, fontweight='bold', pad=15)
ax.set_xlabel(f'PC1 ({exp_var[0]:.1f}%)', fontsize=12)
ax.set_ylabel(f'PC2 ({exp_var[1]:.1f}%)', fontsize=12)


# Remove top and right borders for a cleaner look
ax.spines['top'].set_visible(False)
ax.spines['right'].set_visible(False)

# 6. Build the Custom Multi-Legend
# Because we mapped manually, we must construct the legend manually using Line2D proxy artists
# A. Color Legend (Gut Compartment)
color_handles = [
    mlines.Line2D([], [], color='w', marker='s', markerfacecolor=color_map['Midgut'], markersize=12, label='Midgut'),
    mlines.Line2D([], [], color='w', marker='s', markerfacecolor=color_map['Hindgut'], markersize=12, label='Hindgut')
]
leg_color = ax.legend(handles=color_handles, title='Gut Compartment', bbox_to_anchor=(1.02, 1), loc='upper left', frameon=False, title_fontproperties={'weight':'bold'})
ax.add_artist(leg_color) # Add it to the axes so it isn't overwritten by the next legend

# B. Shape Legend (Timepoint)
shape_handles = [
    mlines.Line2D([], [], color='w', marker=shape_map['1T'], markerfacecolor='grey', markersize=12, label='1T'),
    mlines.Line2D([], [], color='w', marker=shape_map['3T'], markerfacecolor='grey', markersize=12, label='3T')
]
leg_shape = ax.legend(handles=shape_handles, title='Transfer', bbox_to_anchor=(1.02, 0.75), loc='upper left', frameon=False, title_fontproperties={'weight':'bold'})
ax.add_artist(leg_shape)

# C. Fill Legend (Condition)
fill_handles = [
    mlines.Line2D([], [], color='w', marker='o', markerfacecolor='grey', markeredgecolor='grey', markersize=12, label='Procedure 1 (Filled)'),
    mlines.Line2D([], [], color='w', marker='o', markerfacecolor='none', markeredgecolor='grey', markeredgewidth=2, markersize=12, label='Procedure 2 (Unfilled)')
]
leg_fill = ax.legend(handles=fill_handles, title='Procedure', bbox_to_anchor=(1.02, 0.50), loc='upper left', frameon=False, title_fontproperties={'weight':'bold'})

# 7. Show and Save
plt.tight_layout()
plt.savefig('pca_3way_binary.pdf', dpi=300, bbox_inches='tight')
plt.show()
```
